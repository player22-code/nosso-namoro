/* ===== Módulo Mascote do Casal "Juntos" ===== */ let mascoteDados = null; let mascoteCanal = null; function clamp(v, min=0, max=100) { return Math.max(min, Math.min(max, v)); } function atualizarInterfaceMascote() { if (!mascoteDados) return; $('mascote-nome-titulo').textContent = mascoteDados.nome || 'Juntos'; const fome = clamp(mascoteDados.fome); const fel = clamp(mascoteDados.felicidade); const ene = clamp(mascoteDados.energia); const sono = clamp(mascoteDados.sono); const xp = Math.max(0, mascoteDados.experiencia || 0); const nivel = Math.max(1, mascoteDados.nivel || 1); const xpNecessario = nivel * 100; const xpPct = clamp((xp / xpNecessario) * 100); $('val-fome').textContent = fome + '%'; $('barra-fome').style.width = fome + '%'; $('val-felicidade').textContent = fel + '%'; $('barra-felicidade').style.width = fel + '%'; $('val-energia').textContent = ene + '%'; $('barra-energia').style.width = ene + '%'; $('val-sono').textContent = sono + '%'; $('barra-sono').style.width = sono + '%'; $('mascote-nivel').textContent = nivel; $('mascote-xp').textContent = xp + '/' + xpNecessario; $('barra-xp').style.width = xpPct + '%'; // Mensagem contextual do coelho const balao = $('mascote-balao-mensagem'); if (mascoteDados.estado === 'dormindo') { balao.textContent = 'Zzz... Que soninho bom! 😴'; } else if (fome > 70) { balao.textContent = 'Estou com fome! 🥕'; } else if (sono > 70) { balao.textContent = 'Estou com muito sono... 🥱'; } else if (fel > 70) { balao.textContent = 'Estou muito feliz com vocês! 🥰'; } else { balao.textContent = 'Amo passar o tempo com vocês! 💕'; } $('btn-mascote-dormir').textContent = mascoteDados.estado === 'dormindo' ? '☀️ Acordar' : '😴 Dormir'; } async function carregarMascote(storyId) { if (!storyId) return; let { data, error } = await db.from('mascote_casal').select('*').eq('story_id', storyId).maybeSingle(); if (error && error.code !== 'PGRST116') { console.warn('Erro ao buscar mascote:', error); return; } if (!data) { // Criar registro padrão se não existir const novo = { story_id: storyId, nome: 'Juntos', fome: 30, felicidade: 80, energia: 90, sono: 20, experiencia: 0, nivel: 1, estado: 'acordado', updated_by: usuario.id }; const { data: criado, error: errCriar } = await db.from('mascote_casal').insert(novo).select().single(); if (!errCriar) data = criado; } mascoteDados = data; atualizarInterfaceMascote(); } async function salvarAcaoMascote(alteracoes) { if (!mascoteDados || !historiaAtual) return; let novoxp = (mascoteDados.experiencia || 0) + (alteracoes.xp || 0); let novoNivel = mascoteDados.nivel || 1; let xpReq = novoNivel * 100; while (novoxp >= xpReq) { novoxp -= xpReq; novoNivel++; xpReq = novoNivel * 100; } const payload = { fome: clamp(mascoteDados.fome + (alteracoes.fome || 0)), felicidade: clamp(mascoteDados.felicidade + (alteracoes.felicidade || 0)), energia: clamp(mascoteDados.energia + (alteracoes.energia || 0)), sono: clamp(mascoteDados.sono + (alteracoes.sono || 0)), experiencia: novoxp, nivel: novoNivel, estado: alteracoes.estado !== undefined ? alteracoes.estado : mascoteDados.estado, updated_by: usuario.id, updated_at: new Date().toISOString() }; mascoteDados = { ...mascoteDados, ...payload }; atualizarInterfaceMascote(); const { error } = await db.from('mascote_casal').update(payload).eq('id', mascoteDados.id); if (error) console.error('Erro ao atualizar mascote:', error); } function mascoteDesligar() { if (mascoteCanal && db) db.removeChannel(mascoteCanal); mascoteCanal = null; mascoteDados = null; } async function mascoteConectarHistoria() { const storyId = historiaAtual?.historia_id; if (!storyId) { mascoteDesligar(); return; } mascoteDesligar(); await carregarMascote(storyId); mascoteCanal = db.channel('mascote-' + storyId) .on('postgres_changes', { event: '*', schema: 'public', table: 'mascote_casal', filter: 'story_id=eq.' + storyId }, payload => { if (payload.new) { mascoteDados = payload.new; atualizarInterfaceMascote(); } }) .subscribe(); } // Botoes de Ação $('btn-mascote-alimentar').onclick = () => salvarAcaoMascote({ fome: -20, felicidade: 10, xp: 15 }); $('btn-mascote-brincar').onclick = () => salvarAcaoMascote({ felicidade: 20, fome: 15, energia: -15, sono: 15, xp: 20 }); $('btn-mascote-carinho').onclick = () => salvarAcaoMascote({ felicidade: 15, xp: 10 }); $('btn-mascote-dormir').onclick = () => { const novoEstado = mascoteDados?.estado === 'dormindo' ? 'acordado' : 'dormindo'; if (novoEstado === 'dormindo') { salvarAcaoMascote({ energia: 50, sono: -50, estado: 'dormindo', xp: 25 }); } else { salvarAcaoMascote({ estado: 'acordado', xp: 5 }); } };
